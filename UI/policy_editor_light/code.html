# GAVEL OS Production Policy
package gavel.network.production

default allow = false

# Enforce HTTPS for all external traffic
allow {
    input.request.protocol == "https"
    input.request.origin == "external"
    is_trusted_ca(input.request.certificate)
}

is_trusted_ca(cert) {
    ca_list := ["GAVEL_ROOT_01", "SYSTEM_AUTH_X"]
    cert.issuer == ca_list[_]
}

# Deny deprecated cipher suites
deny[msg] {
    ciphers := ["TLS_RSA_WITH_AES_128_CBC_SHA", "TLS_3DES"]
    input.request.cipher == ciphers[_]
    msg := sprintf("Insecure cipher detected: %v", [input.request.cipher])
}
